app.js

The server should always be called app.js.
Here is an example using CommonJS 


        const express = require('express');
        const app = express();

        const PORT = 8080;
        app.listen(PORT, (error) => {
            if (error) {
                console.log("There was an error starting the server", error);
            }
            console.log("Server is running on port", PORT);
        });

Import express with require(), initiate express and save it in a variable. Make it listen to a port.
When using ES Modules, it is done slightly differently, as shown below: 


        import express from "express";
        const app = express();

        const PORT = 8080;
        app.listen(PORT, (error) => {
            if (error) {
                console.log("There was an error starting the server", error);
            }
            console.log("Server is running on port", PORT);
        });
